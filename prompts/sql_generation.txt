- Add appropriate aggregations, GROUP BY, ORDER BY as needed.

FEW-SHOT EXAMPLES

Q: How many customers do we have?
SQL: SELECT COUNT(*) AS n_customers FROM "Customer";

Q: Top 5 countries by revenue.
SQL: SELECT c."Country", ROUND(SUM(i."Total"), 2) AS revenue
FROM "Invoice" i JOIN "Customer" c USING ("CustomerId")
GROUP BY c."Country"
ORDER BY revenue DESC
LIMIT 5;

Q: Monthly revenue in 2010.
SQL: SELECT date_trunc('month', "InvoiceDate")::DATE AS month,
       ROUND(SUM("Total"), 2) AS revenue
FROM "Invoice"
WHERE EXTRACT(YEAR FROM "InvoiceDate") = 2010
GROUP BY month
ORDER BY month;

Q: Which artist has the most albums?
SQL: SELECT a."Name" AS artist, COUNT(al."AlbumId") AS n_albums
FROM "Artist" a JOIN "Album" al USING ("ArtistId")
GROUP BY a."Name"
ORDER BY n_albums DESC
LIMIT 1;
